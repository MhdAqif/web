    const form = document.getElementById("registrationForm");
    const errorBox = document.getElementById("error");
    const successBox = document.getElementById("success");

    function getValue(id) {
        return document.getElementById(id).value.trim();
    }

    function showError(message, id) {
        errorBox.textContent = message;
        successBox.textContent = "";

        if (id) {
            document.getElementById(id).focus();
        }
    }

    form.addEventListener("submit", function(event) {
        event.preventDefault();

        errorBox.textContent = "";
        successBox.textContent = "";

        const fname = getValue("fname");
        const lname = getValue("lname");
        const email = getValue("email");
        const mobile = getValue("mobile");
        const dob = getValue("dob");
        const address = getValue("address");
        const city = getValue("city");
        const pincode = getValue("pincode");
        const state = getValue("state");
        const country = getValue("country");
        const password = getValue("password");
        const tcdate = getValue("tcdate");

        // 1. First name
        const namePattern = /^[A-Za-z]+(?:[ '-][A-Za-z]+)*$/;

        if (!namePattern.test(fname) || fname.length > 30) {
            showError("Enter a valid first name (maximum 30 characters).", "fname");
            return;
        }

        // 2. Last name
        if (!namePattern.test(lname) || lname.length > 30) {
            showError("Enter a valid last name (maximum 30 characters).", "lname");
            return;
        }

        // 3. Email
        const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

        if (!emailPattern.test(email)) {
            showError("Enter a valid email address.", "email");
            return;
        }

        // 4. Date of birth
        if (!dob) {
            showError("Please select your date of birth.", "dob");
            return;
        }

        const today = new Date();
        const birthDate = new Date(dob + "T00:00:00");

        if (birthDate > today) {
            showError("Date of birth cannot be in the future.", "dob");
            return;
        }

        // 5. Mobile number
        if (!/^[0-9]{10}$/.test(mobile)) {
            showError("Mobile number must contain exactly 10 digits.", "mobile");
            return;
        }

        // 6. Gender
        if (!document.querySelector('input[name="gender"]:checked')) {
            showError("Please select your gender.");
            return;
        }

        // 7. Address
        if (address.length < 5 || address.length > 200) {
            showError("Address must contain 5 to 200 characters.", "address");
            return;
        }

        // 8. City
        if (!namePattern.test(city) || city.length > 30) {
            showError("Enter a valid city name (maximum 30 characters).", "city");
            return;
        }

        // 9. PIN code
        if (!/^[0-9]{6}$/.test(pincode)) {
            showError("PIN code must contain exactly 6 digits.", "pincode");
            return;
        }

        // 10. State
        if (!namePattern.test(state)) {
            showError("Enter a valid state name.", "state");
            return;
        }

        // 11. Country
        if (!namePattern.test(country)) {
            showError("Enter a valid country name.", "country");
            return;
        }

        // 12. Hobbies (optional)
        // No validation required.

        // 13. Qualification details
        const boards = document.querySelectorAll(".board");
        const percentages = document.querySelectorAll(".percentage");
        const years = document.querySelectorAll(".year");
        const currentYear = new Date().getFullYear();

        for (let i = 0; i < boards.length; i++) {
            const board = boards[i].value.trim();
            const percentage = percentages[i].value;
            const year = years[i].value;

            // Allow a qualification row to be completely empty.
            if (!board && percentage === "" && year === "") {
                continue;
            }

            // If one value is entered, require all three.
            if (!board || percentage === "" || year === "") {
                showError("Complete all fields in qualification row " + (i + 1) + ".");
                boards[i].focus();
                return;
            }

            if (board.length > 10) {
                showError("Board name must not exceed 10 characters.");
                boards[i].focus();
                return;
            }

            if (!/^\d{1,3}(\.\d{1,2})?$/.test(percentage) ||
                Number(percentage) < 0 || Number(percentage) > 100) {
                showError("Percentage must be between 0 and 100, with at most 2 decimals.");
                percentages[i].focus();
                return;
            }

            if (!/^\d{4}$/.test(year) ||
                Number(year) < 1900 || Number(year) > currentYear) {
                showError("Enter a valid passing year between 1900 and " + currentYear + ".");
                years[i].focus();
                return;
            }
        }

        // 14. Course
        if (!document.querySelector('input[name="course"]:checked')) {
            showError("Please select a course.");
            return;
        }

        // 15. Password
        if (password.length < 8 || password.length > 30) {
            showError("Password must contain 8 to 30 characters.", "password");
            return;
        }

        // 16. Date of transfer certificate (optional)
        if (tcdate && new Date(tcdate + "T00:00:00") > today) {
            showError("Date of TC cannot be in the future.", "tcdate");
            return;
        }

        // All validations passed
        successBox.textContent = "Registration form validated successfully!";
        alert("Registration form validated successfully!");
    });

    form.addEventListener("reset", function() {
        errorBox.textContent = "";
        successBox.textContent = "";
    });
